// 개찰알림 위젯 — Scriptable 앱에 이 글 전체를 붙여 넣어요. (처음 한 번만)
// ▼ 설치 안내 페이지에서 복사하면 아래 두 줄이 자동으로 채워져요. 비어 있으면 앱 설정 탭의 주소·암호를 넣어요
const SERVER = "";   // Apps Script 주소 (https://script.google.com/macros/s/…/exec)
const TOKEN = "";    // 연결 암호

// ---- 아래는 고치지 않아요. 위젯 본체를 인터넷에서 받아 항상 최신으로 유지해요. ----
const fm = module.filename.includes("Documents/iCloud") || module.filename.includes("Mobile Documents") ? FileManager.iCloud() : FileManager.local();
const dir = module.filename.slice(0, module.filename.lastIndexOf("/"));
const core = fm.joinPath(dir, "g2b-alert-widget-core.js");
try {
  const r = new Request("https://friese8807.github.io/g2b/widget-core.js?t=" + Date.now());
  r.timeoutInterval = 15;
  const code = await r.loadString();
  if (code.includes("G2B_ALERT_WIDGET")) fm.writeString(core, code);   // 표식이 있을 때만 저장 (오류 페이지를 저장하지 않게)
} catch (e) {}
if (fm.isFileStoredIniCloud(core) && !fm.isFileDownloaded(core)) await fm.downloadFileFromiCloud(core);
await importModule("g2b-alert-widget-core")({ server: SERVER, token: TOKEN });
